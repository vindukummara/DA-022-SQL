Quick MySQL commands to remember
Operation	Command
Add column	ALTER TABLE table_name ADD column_name datatype;
Modify column	ALTER TABLE table_name MODIFY column_name datatype;
Rename column	ALTER TABLE table_name RENAME COLUMN old TO new;
Drop column	ALTER TABLE table_name DROP COLUMN column_name;
Add PK	        ADD CONSTRAINT ... PRIMARY KEY
Add UNIQUE	ADD CONSTRAINT ... UNIQUE
Add CHECK	ADD CONSTRAINT ... CHECK
Add FK	        ADD CONSTRAINT ... FOREIGN KEY
Drop PK 	DROP PRIMARY KEY
Drop UNIQUE	DROP INDEX constraint_name
Drop FK	        DROP FOREIGN KEY constraint_name
Drop CHECK	DROP CHECK constraint_name
